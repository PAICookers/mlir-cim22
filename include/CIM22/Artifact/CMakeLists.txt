set(CIM22_IMAGE_SCHEMA ${CMAKE_CURRENT_SOURCE_DIR}/CIM22Image.fbs)
set(CIM22_IMAGE_GENERATED_DIR
    ${PROJECT_BINARY_DIR}/include/CIM22/Artifact)
set(CIM22_IMAGE_GENERATED_HEADER
    ${CIM22_IMAGE_GENERATED_DIR}/CIM22Image_generated.h)

add_custom_command(
  OUTPUT ${CIM22_IMAGE_GENERATED_HEADER}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CIM22_IMAGE_GENERATED_DIR}
  COMMAND $<TARGET_FILE:flatbuffers::flatc>
          --cpp
          -o ${CIM22_IMAGE_GENERATED_DIR}
          ${CIM22_IMAGE_SCHEMA}
  DEPENDS ${CIM22_IMAGE_SCHEMA} flatbuffers::flatc
  VERBATIM
)

add_custom_target(MLIRCIM22ArtifactSchemaIncGen
  DEPENDS ${CIM22_IMAGE_GENERATED_HEADER}
)
